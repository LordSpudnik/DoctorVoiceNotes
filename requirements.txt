# ============================================================================
# Doctor Voice Notes - Python dependencies
# ============================================================================
# Target: Python 3.12, Windows 10/11, 64-bit  (PRD Section 9)
#
# Install with:
#   python -m venv venv
#   venv\Scripts\activate
#   pip install -r requirements.txt
#
# NOTE ON VERSIONS:
# The versions below are pinned so the build is reproducible. They were
# current and known-compatible with each other at the time this file was
# written. Before you build the final installer, run:
#   pip list --outdated
# and re-test after any upgrade, especially ctranslate2 <-> faster-whisper,
# since those two must be compatible with each other.
# ============================================================================

# --- Speech recognition (offline) ------------------------------------------
faster-whisper==1.0.3
# faster-whisper depends on ctranslate2 as its inference engine.
# Pinned explicitly here so pip does not silently pull an incompatible major
# version in the future.
ctranslate2==4.4.0

# faster_whisper/utils.py does "import requests" unconditionally but does
# NOT declare requests as a dependency in its own package metadata -
# confirmed by installing faster-whisper alone into an empty venv and
# watching requests NOT get pulled in. Without this, the app fails at
# import time with "No module named 'requests'", even though pip install
# -r requirements.txt reports success with no errors.
requests==2.32.3

# --- Audio capture from the microphone -------------------------------------
sounddevice==0.4.7
# sounddevice needs numpy to hand back audio as arrays.
numpy==1.26.4

# --- Voice activity detection (used to decide when a phrase has ended) ----
# This is NOT in the PRD's dependency list, but it is required to make
# "real-time" transcription behave sensibly (see Phase 3 notes). Without it,
# we would have to guess fixed-length chunks, which cuts words in half.
webrtcvad==2.0.10

# webrtcvad.py calls pkg_resources.get_distribution() at import time.
# setuptools >=81 removed pkg_resources entirely (deprecated, then
# deleted) - confirmed by reproducing in a clean venv: latest setuptools
# installs fine, but "import pkg_resources" then fails outright. Pin
# below that version so pkg_resources still exists to be imported.
setuptools<81

# --- Word document generation ----------------------------------------------
python-docx==1.1.2

# --- Packaging into a Windows executable (see build.spec: onedir mode) -----
pyinstaller==6.10.0

customtkinter==6.0.0

# ============================================================================
# Deliberately EXCLUDED (per PRD Section 4 - Scope - Excluded):
#   - any HTTP client / cloud SDK
#   - any authentication library
#   - any database driver
# requests above is the one exception - it is a hard runtime dependency of
# faster-whisper's own code (see note above), not a deliberate addition of
# networking capability. engine.py never calls anything that uses it
# (local_files_only=True everywhere), it is only imported.
# If you ever see a genuinely new HTTP client, auth library, or database
# driver creep into this file, that is a sign scope is drifting away from
# the "fully offline, single user" design.
# ============================================================================
